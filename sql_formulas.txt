=== ТОП ИГРОКОВ ПО РАЗНЫМ ФОРМУЛАМ ===

================================================================
ФОРМУЛА 1: По винрейту (ТЕКУЩАЯ)
================================================================

SELECT steam_nickname, week_wins, week_games, 
       ROUND(week_wins * 100.0 / week_games, 1) as winrate
FROM users 
WHERE steam_id IS NOT NULL AND week_games > 0
ORDER BY winrate DESC
LIMIT 10


================================================================
ФОРМУЛА 2: По победам
================================================================

SELECT steam_nickname, week_wins, week_games,
       ROUND(week_wins * 100.0 / week_games, 1) as winrate
FROM users 
WHERE steam_id IS NOT NULL AND week_games > 0
ORDER BY week_wins DESC
LIMIT 10


================================================================
ФОРМУЛА 3: ГИБРИД - винрейт × корень(игр)
================================================================

-- PostgreSQL
SELECT steam_nickname, week_wins, week_games,
       ROUND(week_wins * 100.0 / week_games, 1) as winrate,
       ROUND((week_wins * 100.0 / week_games) * SQRT(week_games), 1) as hybrid_score
FROM users 
WHERE steam_id IS NOT NULL AND week_games > 0
ORDER BY hybrid_score DESC
LIMIT 10

-- SQLite (нет SQRT - используем POWER)
SELECT steam_nickname, week_wins, week_games,
       ROUND(week_wins * 100.0 / week_games, 1) as winrate,
       ROUND((week_wins * 100.0 / week_games) * POWER(week_games, 0.5), 1) as hybrid_score
FROM users 
WHERE steam_id IS NOT NULL AND week_games > 0
ORDER BY hybrid_score DESC
LIMIT 10


================================================================
ФОРМУЛА 4: Мин. 5 игр + винрейт
================================================================

SELECT steam_nickname, week_wins, week_games,
       ROUND(week_wins * 100.0 / week_games, 1) as winrate
FROM users 
WHERE steam_id IS NOT NULL AND week_games >= 5
ORDER BY winrate DESC
LIMIT 10


================================================================
ФОРМУЛА 5: ГИБРИД - винрейт × LOG(игр)
================================================================

-- PostgreSQL
SELECT steam_nickname, week_wins, week_games,
       ROUND(week_wins * 100.0 / week_games, 1) as winrate,
       ROUND((week_wins * 100.0 / week_games) * LN(week_games + 1), 1) as hybrid_score
FROM users 
WHERE steam_id IS NOT NULL AND week_games > 0
ORDER BY hybrid_score DESC
LIMIT 10

-- SQLite (нет LN - используем LOG)
SELECT steam_nickname, week_wins, week_games,
       ROUND(week_wins * 100.0 / week_games, 1) as winrate,
       ROUND((week_wins * 100.0 / week_games) * LOG(week_games + 1), 1) as hybrid_score
FROM users 
WHERE steam_id IS NOT NULL AND week_games > 0
ORDER BY hybrid_score DESC
LIMIT 10
